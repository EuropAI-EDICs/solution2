# poc-nspd Implementatieplan (fase 1)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Een nieuwe `poc-nspd/` die planningsvragen grensoverschrijdend voor het North Sea Port District beantwoordt (NL: Vlissingen/Terneuzen/Borsele; BE: haven Gent/Zelzate) met per claim jurisdictie-specifieke normonderbouwing, en de zones toont in een geconfigureerde Leia-viewer-scène.

**Architecture:** Kopie/adapt van de PoC-1-machinerie (`poc/`): deterministische orchestrator over een gecureerd corpus (cite-or-abstain) → formal rules → zone-engine → critic-gates → rapport; nieuw: jurisdictieprofielen (nl/be), jurisdictie-gate (V2b), xborder-merge en `make_viewer_config.py` dat run-output omzet naar een Leia-config (`viewer-data/`). Leia zelf is een git-submodule (`viewer/`), ongepatched.

**Tech Stack:** Python 3.13 (stdlib-first), shapely, pyproj, jsonschema, jinja2; Leia (SvelteKit + Cesium, MIT) via npm/Docker; pytest.

**Spec:** `docs/superpowers/specs/2026-10-14-poc-nspd-design.md` — dit plan implementeert **fase 1** (succes­criterium: grensvergelijkende scan voor wind/zon/energietoets). Fase 2 (overige 10 tracks) krijgt een eigen plan.

## Global Constraints

- `poc/` (Utrecht) blijft onaangeroerd: hergebruik uitsluitend via kopie naar `poc-nspd/`, geen gedeelde refactors.
- Cache-first: een volledige run reproduceert zonder netwerk (fixtures in tests); responses landen onder `poc-nspd/data/cache/`.
- Luid falen: een niet-gepinde bron (lege `serviceUrl`/`typeName`, of registry-status `candidate`) faalt met `ValueError`, nooit stil leeg.
- Cite-or-abstain: alleen geverifieerde citaten worden NormCards (`verified: true`); abstracties of interpretaties zonder citaat → evidence-ledger.
- Jurisdictie-match (V2b): elke normclaim/zone/datasource claim in een run moet een bron-id uit de *eigen* jurisdictie hebben; mismatch = luide faal.
- CRS: brondata NL EPSG:28992 (RD), BE EPSG:31370 (Lambert72); alle tussen- en eindoutputs EPSG:4326 (WGS84). Conversies uitsluitend via pyproj.
- Registry-pinnen (geverifieerd live op 2026-10-14 vanuit deze omgeving, zie taak 1): NL `CVDR709135` (Omgevingsverordening Zeeland, `lokaleregelgeving.overheid.nl`); 3D BAG `https://data.3dbag.nl/v20250903/cesium3dtiles/lod22/tileset.json`; BE `https://geo.api.vlaanderen.be/GRB/wfs` (HTTP 200), `https://geo.api.vlaanderen.be/DHMV/wms` (HTTP 200, laag `DHMV-II`), `https://www.mercator.vlaanderen.be/raadpleegdienstenmercatorpubliek/ows` (HTTP 200, SBZ-lagen `nz:nz_n_gevoelige_habitats_sbz_2025`). De oude host `geoservices.informatievlaanderen.be` bestaat niet meer (geen DNS) — nooit gebruiken.
- Geen LLM standaard aan: alle LLM-seams van PoC-1 blijven propose-only en uit (lokaal Ollama, expliciet env-vlag).
- Test-suite offline: standaard `pytest` draait de snelle suite; `golden`-marker expliciet (`pytest -m golden`), zoals `poc/pytest.ini`.
- Commit per taak; commit-bericht in het Nederlands, conventie `feat(poc-nspd): …`.

`poc-nspd/bbox.py`:

```python
"""NSPD-werkkader: één WGS84-bbox, afgeleide RD- en Lambert72-kaders.

Spec: docs/superpowers/specs/2026-10-14-poc-nspd-design.md §4.3.
Gepind in taak 1 van plan 2026-10-14-poc-nspd.
"""
from __future__ import annotations

from functools import lru_cache
from pyproj import Transformer

NSPD_BBOX_WGS84: tuple[float, float, float, float] = (3.45, 51.02, 3.88, 51.48)
# kader: Vlissingen/Borsele (NW) → haven Gent/Zelzate (ZO)


def _transform(bbox: tuple[float, float, float, float], to_epsg: int) -> tuple:
    x0, y0, x1, y1 = bbox
    t = Transformer.from_crs("EPSG:4326", f"EPSG:{to_epsg}", always_xy=True)
    corners = [t.transform(x, y) for x in (x0, x1) for y in (y0, y1)]
    xs = [c[0] for c in corners]
    ys = [c[1] for c in corners]
    return (min(xs), min(ys), max(xs), max(ys))


@lru_cache(maxsize=1)
def nspd_bbox_rd() -> tuple[float, float, float, float]:
    return _transform(NSPD_BBOX_WGS84, 28992)


@lru_cache(maxsize=1)
def nspd_bbox_l72() -> tuple[float, float, float, float]:
    return _transform(NSPD_BBOX_WGS84, 31370)
```

```python
"""Live-probe van de BE-diensten (taak 3/5): GetCapabilities + typeNames.

Schrijft data/probe_<datum>.json; is GEEN pytest (netwerk). Discipline:
laag die niet live antwoordt blijft status=candidate in sources.json.
Run: python3 data/probe_be_sources.py
"""
from __future__ import annotations

import json
import re
import urllib.request
from datetime import datetime, timezone
from pathlib import Path

TARGETS = {
    "geopunt-grb-wfs": "https://geo.api.vlaanderen.be/GRB/wfs?service=WFS&version=2.0.0&request=GetCapabilities",
    "geopunt-dhmv-wms": "https://geo.api.vlaanderen.be/DHMV/wms?service=WMS&request=GetCapabilities",
    "mercator-sbz-wms": "https://www.mercator.vlaanderen.be/raadpleegdienstenmercatorpubliek/ows?service=WMS&request=GetCapabilities",
    "mercator-ruimtelijk-beleid": "https://www.mercator.vlaanderen.be/raadpleegdienstenmercatorpubliek/ows?service=WFS&version=2.0.0&request=GetCapabilities",
}

INTEREST = re.compile(r"(?i)(adp|gebouw|gewestplan|rup|ruimtelijk|sbz|natura)")


def _probe(url: str) -> dict:
    try:
        req = urllib.request.Request(url, headers={"User-Agent": "nLDT-toolbox-poc-nspd/1.0"})
        with urllib.request.urlopen(req, timeout=60) as r:
            body = r.read().decode("utf-8", errors="replace")
            status = r.status
    except Exception as exc:  # luid in het probe-rapport, faal-status voor de registry
        return {"url": url, "status": None, "error": f"{type(exc).__name__}: {exc}"}
    names = sorted(set(re.findall(r"<Name>([^<]{3,90})</Name>", body)))
    return {
        "url": url,
        "status": status,
        "bytes": len(body),
        "interestingNames": [n for n in names if INTEREST.search(n)][:60],
    }


def main() -> None:
    out = Path(__file__).resolve().parent / (
        "probe_" + datetime.now(timezone.utc).strftime("%Y-%m-%d") + ".json")
    report = {
        "generatedAt": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "probes": {sid: _probe(url) for sid, url in TARGETS.items()},
    }
    out.write_text(json.dumps(report, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"geschreven: {out}")
    for sid, p in report["probes"].items():
        print(f"  {sid}: HTTP {p['status']}, {len(p.get('interestingNames', []))} interessante namen")


if __name__ == "__main__":
    main()
```

    *,
    service_url: str,
    type_name: str,
    bbox_l72: tuple[float, float, float, float],
    cache_name: str,
    source_id: str,
    refresh: bool = False,
    transport: Transport | None = None,
) -> dict:

    Cache-first onder CACHE/be/<cache_name>; niet-gepinde laag faalt luid.

            f"(zie de note bij deze bron in poc-nspd/data/sources.json)")

"""Evidence-extractor: kandidaatcitaten uit corpus-tekst (deterministisch).

Curation-flow: extract_candidates() → curator verifieert en promote tot
evidence-*.json (cite-or-abstain: alleen gecontroleerde citaten verder).
"""
from __future__ import annotations

import re
from datetime import datetime, timezone

_ARTIKEL_RE = re.compile(r"^\s*Artikel\s+([0-9]+(?:\.[0-9]+)*)\b", re.M)
_EVID_PREFIX = {"wind": "EVW", "zon": "EVZ", "energietoets": "EVE"}
_PARA_SCHUIN = re.compile(r"^\s{2,}\d+\s")


def _artikelgrenzen(tekst: str) -> list[tuple[str, int, int]]:
    koppen = list(_ARTIKEL_RE.finditer(tekst))
    grenzen = []
    for i, m in enumerate(koppen):
        start = m.start()
        einde = koppen[i + 1].start() if i + 1 < len(koppen) else len(tekst)
        grenzen.append((f"Artikel {m.group(1)}", start, einde))
    return grenzen


def extract_candidates(
    *,
    tekst: str,
    track: str,
    trefwoorden: tuple[str, ...],
    jurisdictie: str,
    source_url: str,
) -> list[dict]:
    ev_prefix = _EVID_PREFIX[track]
    kandidaten: list[dict] = []
    for kop, start, einde in _artikelgrenzen(tekst):
        blok = tekst[start:einde]
        laag = blok.lower()
        if any(tw.lower() in laag for tw in trefwoorden):
            quote = " ".join(blok.split())[:600]
            kandidaten.append({
                "id": f"{ev_prefix}-{len(kandidaten) + 1:02d}",
                "jurisdictie": jurisdictie,
                "track": track,
                "locator": kop,
                "quote": quote,
                "sourceUrl": source_url,
                "retrieved": datetime.now(timezone.utc).isoformat(timespec="seconds"),
                "verified": False,  # curator-promotie zet deze op true
            })
    return kandidaten

```python
"""V2b: jurisdictie-match — geen claim over de grens (spec §4.6)."""
from __future__ import annotations

from typing import Any, Mapping, Sequence

_BE_NL_TYPES = {
    "nl": {"omgevingsverordening-nl", "omgevingsvisie-nl"},
    "be": {"rup-be", "gewestplan-be", "vcro-be", "vlarem-be"},
}


class JurisdictieError(RuntimeError):
    pass


def run_v2b(
    *,
    normcards: Sequence[Mapping[str, Any]],
    formal_rules: Sequence[Mapping[str, Any]],
    sources_registry: Mapping[str, Any],
    jurisdictie: str,
    strict: bool = False,
) -> dict:
    reg = sources_registry.get("sources", [])
    bron_jur: dict[str, str] = {s.get("id", ""): s.get("jurisdictie", "") for s in reg}
    checks: list[dict] = []
    for k in normcards:
        if k.get("jurisdictie") != jurisdictie:
            checks.append({"id": "V2B-KAART", "status": "fail",
                           "detail": f"{k.get('id')}: jurisdictie {k.get('jurisdictie')} in run {jurisdictie}"})
        itype = k.get("instrumentType", "")
        if itype and itype not in _BE_NL_TYPES[jurisdictie]:
            checks.append({"id": "V2B-INSTRUMENT", "status": "fail",
                           "detail": f"{k.get('id')}: instrumentType {itype} niet thuis in {jurisdictie}"})
    for fr in formal_rules:
        if fr.get("jurisdictie") != jurisdictie:
            checks.append({"id": "V2B-RULE", "status": "fail",
                           "detail": f"{fr.get('id')}: jurisdictie {fr.get('jurisdictie')} in run {jurisdictie}"})
        kaart = next((k for k in normcards if k.get("id") == fr.get("normCardId")), None)
        if kaart and kaart.get("jurisdictie") != jurisdictie:
            checks.append({"id": "V2B-RULEKAART", "status": "fail",
                           "detail": f"{fr.get('id')}: linkt naar {kaart.get('id')} ({kaart.get('jurisdictie')})"})
    status = "pass" if not any(c["status"] == "fail" for c in checks) else "fail"
    if status == "fail" and strict:
        raise JurisdictieError("; ".join(c["detail"] for c in checks if c["status"] == "fail"))
    return {"status": status, "checks": checks}
```

```python
"""Grensvergelijkende merge + rapport (spec §4.4)."""
from __future__ import annotations

import json
from pathlib import Path
from typing import Any, Mapping

from jinja2 import Environment, FileSystemLoader

_TEMPLATE_DIR = Path(__file__).resolve().parent / "templates"


def merge_xborder(nl_run: Mapping[str, Any], be_run: Mapping[str, Any], *, track: str) -> dict:
    zones = {"type": "FeatureCollection", "features": []}
    rijen: list[dict] = []
    for run in (nl_run, be_run):
        land = run["land"]
        fc = run.get("zones") or {"type": "FeatureCollection", "features": []}
        for feat in fc.get("features", []):
            props = dict(feat.get("properties", {}))
            props["land"] = land
            zones["features"].append({**feat, "properties": props})
        for rij in run.get("decisionRows", []):
            rijen.append({**rij, "land": land})
    return {"track": track, "zones": zones, "decisionRows": rijen}


def write_xborder_report(out_dir: Path, merged: Mapping[str, Any], *, track: str) -> Path:
    out_dir.mkdir(parents=True, exist_ok=True)
    geo_path = out_dir / f"zones-{track}-nspd.geojson"
    geo_path.write_text(json.dumps(merged["zones"], ensure_ascii=False), encoding="utf-8")
    env = Environment(loader=FileSystemLoader(str(_TEMPLATE_DIR)), autoescape=True)
    html = env.get_template("xborder-report.html.j2").render(
        track=track, rows=merged["decisionRows"], geojsonName=geo_path.name)
    pad = out_dir / "report-xborder.html"
    pad.write_text(html, encoding="utf-8")
    return pad
```

```python
import json
import os
import sys
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

pytestmark = pytest.mark.golden

TRACKS = ("wind-nl", "wind-be", "zon-nl", "zon-be")


def _run(track: str, tmp: Path) -> dict:
    import subprocess
    uit = tmp / track
    subprocess.run(
        [sys.executable, str(ROOT / "run.py"), "--use-case", track, "--out-dir", str(uit),
         "--offline", "--refresh=False"],
        check=True, cwd=ROOT, env={**os.environ, "PYTHONPATH": str(ROOT)},
    )
    return json.loads((uit / "run_summary.json").read_text(encoding="utf-8"))


@pytest.mark.parametrize("track", TRACKS)
def test_golden_track(track, tmp_path):
    summary = _run(track, tmp_path)
    golden = ROOT / "pipeline" / "goldens-nspd" / f"{track}.json"
    if not golden.is_file():  # eerste keer: freeze
        golden.parent.mkdir(parents=True, exist_ok=True)
        golden.write_text(json.dumps(summary, indent=1, sort_keys=True), encoding="utf-8")
        return
    vast = json.loads(golden.read_text(encoding="utf-8"))
    for veld in ("verdict", "output_rules", "formalized", "rejected"):
        assert summary.get(veld) == vast.get(veld), f"{track}: {veld} gedrift"
    assert summary.get("v2b", {}).get("status") == "pass"
```

Run: `cd poc-nspd && python3 -m pytest -m golden -v`
Expected: eerste run schrijft goldens (4 bestanden); tweede run PASS. Vlaggen die `run.py` indien nodig krijgt (voeg toe in `build_parser` en verwerk in `main()`):
