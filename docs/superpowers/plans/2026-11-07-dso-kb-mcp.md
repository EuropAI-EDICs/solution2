# DSO-kennisbank via MCP — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** De DSO-kennisbank (`dso_overlap/out/`) ontsluiten voor toekomstige agents via negen OGC-processen achter een nieuwe MCP-façadeserver `nldt-dso-mcp` op :8094.

**Architecture:** Aanpak 2 uit de spec — pure lees-/refresh-logica in `services/dso_kb.py` (pad-injecteerbaar, `poc_readops`-patroon), dunne proceswrappers in `services/process_adapter/dso_handlers.py` (registratie in `handlers.py`, jobId/status/prov via de bestaande job-store), en een façade `services/mcp_servers/dso_server.py` die spiegelt met `dso_tools.py`. Cache-first: alleen `dso-kb-refresh` gaat live (subprocess naar de bestaande `compare`/`tpod` CLIs), keys blijven server-side in `.env`.

**Tech Stack:** Python 3.13 in `nldt/.venv` (heeft `mcp` SDK 2.x, `pytest`, `yaml`, `jsonschema`, `shapely`), MCP SDK 2.x `MCPServer` + `@mcp.tool()`, FastAPI/uvicorn via bestaand `http_transport.py`, stdlib `json`/`pathlib`/`subprocess`.

**Spec:** `docs/superpowers/specs/2026-11-07-dso-kb-mcp-design.md` (commit `cda7769`).

## Global Constraints

- Poort MCP-server: **8094**; transport/bearer identiek aan :8090–8093 (`NLDT_MCP_TRANSPORT=streamable-http`, `headers.py`/`require_bearer` via `http_transport.run_mcp_http`).
- Proces-ids: kebab-case (`dso-kb-*`), MCP-toolnamen snake_case (`dso_kb_*`); invoernamen camelCase (`wIdPrefix`, `metTpod`) conform poc-patroon.
- Uitvoerbegrenzing: default 100 rijen, hard maximum 500 (`LIMIT_DEFAULT`/`LIMIT_MAX`).
- DSO_API_KEY/RP_API_KEY blijven in de gitignored `.env`; geen enkele functie geeft keys of headers terug. Live gaat alléén `ververs()` (subprocess naar de bestaande CLIs — die lezen `.env` zelf).
- `NLDT_OFFLINE=1`: leesprocessen werken, `ververs()` weigert met duidelijke melding.
- Geen LLM in deze dienst; geen mutaties buiten `dso_overlap/out/`; nieuwe locaties/instrumenten blijven buiten MCP.
- Bestandsnamen cache: `{instrument_id}__{location_id}.{kind}.json`; `.annotaties.meta.json`/`.documentstructuur.meta.json` bevatten `sha256`.
- Tests: `nldt/tests/`, draaien met `nldt/.venv/bin/python -m pytest nldt/tests/<bestand> -v` vanuit de repo-root (`nldt/conftest.py` zet `nldt/` op `sys.path`).
- Commits: conventional commits, Nederlandse beschrijving, alléén de bestanden van de taak.

```python
"""Pure lees- en refresh-laag over de DSO-kennisbank (dso_overlap/out).

De dienst leest het bestandssysteem — de agent nooit direct. Alle functies
zijn puur en via ``out_dir``/``instrumenten``/``locaties`` injecteerbaar
(het poc_readops-testpatroon). Alleen :func:`ververs` schrijft, en alléén
naar de gitignored cache; live sleutels blijven server-side in ``.env``.
Ontwerp: docs/superpowers/specs/2026-11-07-dso-kb-mcp-design.md.
"""
from __future__ import annotations

import json
import os
import re
import subprocess
import sys
from pathlib import Path
from typing import Any, Callable

NLDT_ROOT = Path(__file__).resolve().parents[1]
WORKSPACE = NLDT_ROOT.parent
DEFAULT_OUT = NLDT_ROOT / "simulation" / "regelrecht" / "dso_overlap" / "out"

LIMIT_DEFAULT = 100
LIMIT_MAX = 500

_ARTIKEL_RE = re.compile(r"__art_([\d.]+)")


class DsoKbFout(RuntimeError):
    """Verwachte fout met de geldige opties in de melding."""


def _laat_instrumenten() -> tuple[dict[str, dict[str, Any]], dict[str, dict[str, Any]]]:
    """INSTRUMENTS + locaties uit dso_overlap/instruments.py (lazy import)."""
    root = str(WORKSPACE)
    if root not in sys.path:
        sys.path.insert(0, root)
    from nldt.simulation.regelrecht.dso_overlap.instruments import (
        INSTRUMENTS,
        load_locations,
    )

    return INSTRUMENTS, load_locations()


def _limiet(limiet: int | None) -> int:
    if limiet is None:
        return LIMIT_DEFAULT
    if limiet < 1:
        raise DsoKbFout(f"limiet moet >= 1 zijn (kreeg {limiet})")
    return min(limiet, LIMIT_MAX)


def _resolve_paar(
    instrument: str,
    locatie: str | None,
    instrumenten: dict[str, dict[str, Any]] | None,
    locaties: dict[str, dict[str, Any]] | None,
) -> tuple[dict[str, Any], str, str]:
    if instrumenten is None or locaties is None:
        instrumenten, locaties = _laat_instrumenten()
    if instrument not in instrumenten:
        raise DsoKbFout(
            f"onbekend instrument {instrument!r}; geldig: {sorted(instrumenten)}"
        )
    meta = instrumenten[instrument]
    lid = locatie or meta.get("default_location_id")
    if lid not in locaties:
        raise DsoKbFout(f"onbekende locatie {lid!r}; geldig: {sorted(locaties)}")
    return meta, lid, f"{instrument}__{lid}"


def _lees_json(pad: Path) -> Any:
    if not pad.is_file():
        raise DsoKbFout(
            f"artefact ontbreekt: {pad.name} — draai eerst dso-kb-refresh (of de CLI)"
        )
    return json.loads(pad.read_text(encoding="utf-8"))


def _kort(vol: str | None) -> str | None:
    return vol[:16] if vol else None


def lees_annotaties(
    instrument: str,
    locatie: str | None = None,
    artikel: str | None = None,
    detail: bool = False,
    limiet: int | None = None,
    out_dir: Path | None = None,
    instrumenten: dict[str, dict[str, Any]] | None = None,
    locaties: dict[str, dict[str, Any]] | None = None,
) -> dict[str, Any]:
    """Regeltekstannotaties per artikelnummer, gebonden en filterbaar."""
    out = out_dir or DEFAULT_OUT
    _meta, lid, stem = _resolve_paar(instrument, locatie, instrumenten, locaties)
    data = _lees_json(out / f"{stem}.annotaties.json")
    grens = _limiet(limiet)

    wId_artikel: dict[str, str] = {}
    rijen: dict[str, dict[str, Any]] = {}
    for rt in data.get("regelteksten") or []:
        wid = rt.get("wId", "")
        m = _ARTIKEL_RE.search(wid)
        num = m.group(1) if m else wid
        wId_artikel[wid] = num
        rij = rijen.setdefault(
            num, {"artikel": num, "wIds": [], "regelteksten": 0, "activiteiten": [], "regelIds": []}
        )
        rij["regelteksten"] += 1
        if len(rij["wIds"]) < LIMIT_MAX:
            rij["wIds"].append(wid)

    for regel in (data.get("regelsVoorIedereen") or []) + (data.get("instructieregels") or []):
        num = wId_artikel.get(regel.get("regeltekstRef", ""))
        if num is None:
            continue
        rij = rijen[num]
        if len(rij["regelIds"]) < LIMIT_MAX:
            rij["regelIds"].append(regel.get("identificatie"))
        for ala in regel.get("activiteitLocatieaanduidingen") or []:
            act = (ala.get("activiteitRef") or "").rsplit(".", 1)[-1]
            if act and act not in rij["activiteiten"] and len(rij["activiteiten"]) < LIMIT_MAX:
                rij["activiteiten"].append(act)

    alle = sorted(rijen.values(), key=lambda r: r["artikel"])
    if artikel is not None:
        alle = [r for r in alle if r["artikel"] == artikel]
        if not alle:
            raise DsoKbFout(
                f"artikel {artikel!r} niet in DSO-hit; geldig (eerste {grens}): "
                f"{[r['artikel'] for r in sorted(rijen.values(), key=lambda r: r['artikel'])][:grens]}"
            )

    vervolg = None
    if len(alle) > grens:
        alle = alle[:grens]
        vervolg = {"aanwijzing": "verfijn met artikel= of verhoog limiet", "maxLimiet": LIMIT_MAX}

    uit: dict[str, Any] = {
        "instrument": instrument,
        "locatie": lid,
        "totaalArtikelen": len(rijen),
        "counts": {
            "regelteksten": len(data.get("regelteksten") or []),
            "regelsVoorIedereen": len(data.get("regelsVoorIedereen") or []),
            "instructieregels": len(data.get("instructieregels") or []),
            "locaties": len(data.get("locaties") or []),
            "gebiedsaanwijzingen": len(data.get("gebiedsaanwijzingen") or []),
            "activiteiten": len(data.get("activiteiten") or []),
        },
        "artikelen": alle,
        "vervolg": vervolg,
    }
    if detail and alle:
        wids = set(alle[0]["wIds"])
        uit["regelteksten"] = [
            rt for rt in (data.get("regelteksten") or []) if rt.get("wId") in wids
        ][:LIMIT_MAX]
    return uit
```

```python
def _vlak_componenten(
    componenten: list[Any] | None, diepte: int, uit: list[dict[str, Any]]
) -> None:
    for c in componenten or []:
        uit.append(
            {"diepte": diepte, "soort": c.get("soort"), "wId": c.get("wId"), "titel": c.get("titel")}
        )
        _vlak_componenten((c.get("_embedded") or {}).get("documentComponenten"), diepte + 1, uit)


def lees_documentstructuur(
    instrument: str,
    locatie: str | None = None,
    wId_prefix: str | None = None,
    limiet: int | None = None,
    out_dir: Path | None = None,
    instrumenten: dict[str, dict[str, Any]] | None = None,
    locaties: dict[str, dict[str, Any]] | None = None,
) -> dict[str, Any]:
    """Documentstructuur als vlakke, gebonden componentreeks."""
    out = out_dir or DEFAULT_OUT
    _meta, lid, stem = _resolve_paar(instrument, locatie, instrumenten, locaties)
    data = _lees_json(out / f"{stem}.documentstructuur.json")
    vlak: list[dict[str, Any]] = []
    _vlak_componenten((data.get("_embedded") or {}).get("documentComponenten"), 0, vlak)
    totaal = len(vlak)
    if wId_prefix:
        vlak = [c for c in vlak if str(c.get("wId") or "").startswith(wId_prefix)]
    grens = _limiet(limiet)
    vervolg = None
    if len(vlak) > grens:
        vlak = vlak[:grens]
        vervolg = {"aanwijzing": "verfijn met wIdPrefix= of verhoog limiet", "maxLimiet": LIMIT_MAX}
    return {
        "instrument": instrument,
        "locatie": lid,
        "totaalComponenten": totaal,
        "totaalNaFilter": len(vlak),
        "componenten": vlak,
        "vervolg": vervolg,
    }
```

```python
from urllib.parse import parse_qsl, urlsplit  # bovenin bij de imports

_DATUM_PARAMS = ("geldigOp", "inWerkingOp", "beschikbaarOp")


def _normaliseer(obj: Any) -> Any:
    """Recursief volgorde-onafhankelijk + datumecho-vrij (spec §5)."""
    if isinstance(obj, dict):
        return {k: _normaliseer(obj[k]) for k in sorted(obj)}
    if isinstance(obj, list):
        items = [_normaliseer(x) for x in obj]
        return sorted(items, key=lambda x: json.dumps(x, sort_keys=True, ensure_ascii=False))
    if isinstance(obj, str) and obj.startswith("http"):
        onderdelen = urlsplit(obj)
        query = parse_qsl(onderdelen.query, keep_blank_values=True)
        gefilterd = [(k, v) for k, v in query if k not in _DATUM_PARAMS]
        if len(gefilterd) != len(query):
            from urllib.parse import urlencode, urlunsplit

            return urlunsplit(
                (onderdelen.scheme, onderdelen.netloc, onderdelen.path, urlencode(gefilterd), onderdelen.fragment)
            )
    return obj


def _regel_multiset(annotaties: dict[str, Any]) -> dict[str, list[tuple]]:
    uit: dict[str, list[tuple]] = {}
    for veld in ("regelsVoorIedereen", "instructieregels"):
        for regel in annotaties.get(veld) or []:
            ident = regel.get("identificatie", "")
            tupels = sorted(
                (
                    ala.get("activiteitRef") or "",
                    tuple(sorted(ala.get("locatieRefs") or [])),
                    ((ala.get("symbolisatie") or {}).get("symboolcode") or ""),
                )
                for ala in regel.get("activiteitLocatieaanduidingen") or []
            )
            uit.setdefault(ident, []).extend(tupels)
    return uit


def _per_identificatie(lijst: list[Any] | None) -> dict[str, Any]:
    return {
        item.get("identificatie", ""): _normaliseer(item) for item in (lijst or [])
    }

    """Canonieke (volgorde- en datumecho-onafhankelijke) vergelijking van annotatie-responsies."""
    categorieen: dict[str, Any] = {}

    wids_o = {r.get("wId") for r in oud.get("regelteksten") or []}
    wids_n = {r.get("wId") for r in nieuw.get("regelteksten") or []}
    if wids_o != wids_n:
        categorieen["regelteksten"] = {
            "erbij": sorted(wids_n - wids_o),
            "weg": sorted(wids_o - wids_n),
        }

        anders = sorted(k for k in set(rec_o) | set(rec_n) if rec_o.get(k) != rec_n.get(k))
        if anders:
            categorieen[naam] = {"gewijzigd": anders}
```

```python
def _subprocess_run(args: list[str]) -> None:
    proc = subprocess.run(args, cwd=WORKSPACE, capture_output=True, text=True)
    if proc.returncode != 0:
        staart = (proc.stderr or proc.stdout or "").strip()[-400:]
        raise DsoKbFout(f"refresh mislukte ({args[2]}): {staart}")


def ververs(
    instrument: str,
    locatie: str | None = None,
    met_tpod: bool = True,
    out_dir: Path | None = None,
    runner: Callable[[list[str]], None] | None = None,
    instrumenten: dict[str, dict[str, Any]] | None = None,
    locaties: dict[str, dict[str, Any]] | None = None,
) -> dict[str, Any]:
    """Live PRE-refresh via de bestaande compare/tpod-CLIs + canonieke drift-uitslag.

    Enige functie met netwerk; keys blijven in .env (de CLI's lezen die zelf).
    Runner is injecteerbaar voor tests.
    """
    if os.environ.get("NLDT_OFFLINE") == "1":
        raise DsoKbFout(
            "NLDT_OFFLINE=1 — live refresh uitgeschakeld; de leesprocessen werken gewoon"
        )
    out = out_dir or DEFAULT_OUT
    _meta, lid, stem = _resolve_paar(instrument, locatie, instrumenten, locaties)
    run = runner or _subprocess_run

    ann_pad = out / f"{stem}.annotaties.json"
    oud_ann = json.loads(ann_pad.read_text(encoding="utf-8")) if ann_pad.is_file() else None
    sha_oud = None
    meta_pad = out / f"{stem}.annotaties.meta.json"
    if meta_pad.is_file():
        sha_oud = _kort(json.loads(meta_pad.read_text(encoding="utf-8")).get("sha256"))

    args = [sys.executable, "-m", "nldt.simulation.regelrecht.dso_overlap.compare",
            "--instruments", instrument]
    if locatie:
        args += ["--location", locatie]
    run(args)

    herbouwd = False
    if met_tpod:
        run([sys.executable, "-m", "nldt.simulation.regelrecht.tpod",
             "--instruments", instrument, "--toets", "--geo"])
        herbouwd = True

    nieuw_ann = json.loads(ann_pad.read_text(encoding="utf-8")) if ann_pad.is_file() else None
    sha_nieuw = None
    if meta_pad.is_file():
        sha_nieuw = _kort(json.loads(meta_pad.read_text(encoding="utf-8")).get("sha256"))

    drift = (
        canonieke_drift(oud_ann, nieuw_ann)
        if oud_ann is not None and nieuw_ann is not None
        else {"inhoudelijkeDrift": "onbekend", "categorieen": {}, "reden": "geen oude versie in cache"}
    )
    return {
        "instrument": instrument,
        "locatie": lid,
        "annShaOud": sha_oud,
        "annShaNieuw": sha_nieuw,
        "drift": drift,
        "tpod": {"herbouwd": herbouwd},
```

```python
from __future__ import annotations
```

```python
import os
from typing import Any

import httpx
from mcp.server.mcpserver import MCPServer

from services.mcp_servers.client import PROCESS_URL
from services.mcp_servers.headers import mcp_auth_headers
from services.mcp_servers.dso_tools import DSO_TOOLS

mcp = MCPServer("nldt-dso-mcp")


async def _execute_remote(process_id: str, inputs: dict[str, Any]) -> dict[str, Any]:
    async with httpx.AsyncClient(timeout=300.0) as client:
        resp = await client.post(
            f"{PROCESS_URL.rstrip('/')}/processes/{process_id}/execution",
            json={"inputs": inputs, "backend": "local"},
            headers=mcp_auth_headers(),
        )
        resp.raise_for_status()
        return resp.json()


def _execute_local(process_id: str, inputs: dict[str, Any]) -> dict[str, Any]:
    from services.process_adapter import jobs as job_store

    return job_store.create_job(process_id, inputs, backend="local")


async def _run_dso_tool(tool_name: str, arguments: dict[str, Any]) -> str:
    spec = DSO_TOOLS.get(tool_name)
    if spec is None:
        raise ValueError(f"Unknown tool: {tool_name}")
    inputs = {k: v for k, v in arguments.items() if v is not None}
    if os.environ.get("NLDT_OFFLINE") == "1":
        result = _execute_local(spec["process_id"], inputs)
    else:
        result = await _execute_remote(spec["process_id"], inputs)
    payload = {
        "tool": tool_name,
        "processId": spec["process_id"],
        "jobId": result.get("jobId"),
        "status": result.get("status"),
        "outputs": result.get("outputs"),
        "prov": result.get("prov"),
    }
    return json.dumps(payload, indent=2, default=str)
```
